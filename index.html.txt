    <script>
        // Lucide İkonları Yükle
        lucide.createIcons();

        // Chart.js Grafik Kurulumu
        const ctx = document.getElementById('priceChart').getContext('2d');
        
        // Örnek Grafik Verileri
        const labels = ['Ocak', 'Şubat', 'Mart', 'Nisan', 'Mayıs', 'Haziran', 'Temmuz', 'Ağustos'];
        const priceData = [240, 255, 248, 275, 290, 282, 305, 312.5];

        let priceChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'Fiyat (TL)',
                    data: priceData,
                    borderColor: '#3B82F6',
                    backgroundColor: 'rgba(59, 130, 246, 0.1)',
                    borderWidth: 2,
                    fill: true,
                    tension: 0.3,
                    pointRadius: 4,
                    pointBackgroundColor: '#3B82F6'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { display: false }
                },
                scales: {
                    x: {
                        grid: { color: 'rgba(255, 255, 255, 0.05)' },
                        ticks: { color: '#94A3B8' }
                    },
                    y: {
                        grid: { color: 'rgba(255, 255, 255, 0.05)' },
                        ticks: { color: '#94A3B8' }
                    }
                }
            }
        });

        // Arama/Güncelleme Fonksiyonu Simülasyonu
        function updateDashboard() {
            const symbol = document.getElementById('symbolInput').value.toUpperCase();
            if(!symbol) return;
            
            document.getElementById('stockTitle').innerText = `${symbol} - Analiz Paneli`;
            document.getElementById('stockPrice').innerText = `${(Math.random() * 200 + 50).toFixed(2)} TL`;
            
            // Grafiğe rastgele veri yükle (Simülasyon)
            priceChart.data.datasets[0].data = Array.from({length: 8}, () => Math.floor(Math.random() * 100) + 200);
            priceChart.update();
        }
    </script>
